function shapeNoise(distance, params) {
  if (params.shaping === 'Threshold')
    return distance <= params.threshold ? 1 : 0;

  if (params.shaping === 'Invert Threshold')
    return distance > params.invertThreshold ? 1 : 0;

  const fromBand = Math.abs(distance - 0.5);
  return fromBand <= params.bandWidth / 2 ? 1 : 0;
}
